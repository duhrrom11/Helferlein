async function speichereInCloud() { const alleTermine = sammleTermineAusAllenTools(); const payload = { erstelltAm: new Date().toISOString(), events: alleTermine }; try { let endpoint = NPOINT_API_BASE; let method = 'POST'; // Wenn wir bereits eine echte ID haben, aktualisieren wir diese (PUT) if (hubBinId && !hubBinId.startsWith('hub_')) { endpoint = NPOINT_API_BASE + hubBinId; method = 'PUT'; } let response = await fetch(endpoint, { method: method, headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' }, body: JSON.stringify(payload) }); if (response.ok) { const result = await response.json(); const newId = result.id || result.binId; if (newId) { hubBinId = newId; localStorage.setItem('active_hub_bin_id', hubBinId); window.history.replaceState({}, '', window.location.pathname + "?id=" + hubBinId); alert("✅ Erfolgreich in der npoint.io Cloud gespeichert! ID: " + hubBinId); await ladeDatenAusCloud(); return; } } // Falls npoint.io den POST blockiert, legen wir den Container direkt an throw new Error("npoint.io hat den Request abgewiesen."); } catch (e) { console.warn("Cloud-Direktspeicherung fehlgeschlagen, initialisiere lokalen Fallback-Container.", e); // Falls noch keine echte ID da ist, erzeugen wir den lokalen Container if (!hubBinId || hubBinId.startsWith('hub_')) { hubBinId = "hub_" + Math.random().toString(36).substring(2, 9); localStorage.setItem('active_hub_bin_id', hubBinId); window.history.replaceState({}, '', window.location.pathname + "?id=" + hubBinId); } localStorage.setItem(hubBinId, JSON.stringify(payload)); alert("✅ Erfolgreich lokal im Browser-Container gesichert: " + hubBinId + "\n(Da npoint.io im direkten Root-POST blockiert, läuft das Hub lokal und synchronisiert sich mit deinen Tools)."); sammleUndZeigeTermine({ events: alleTermine }); } }
