async function speichereInCloud() { const alleTermine = sammleTermineAusAllenTools(); const payload = { erstelltAm: new Date().toISOString(), events: alleTermine }; try { let endpoint = NPOINT_API_BASE; let method = 'POST'; // Falls wir bereits eine echte ID haben, nutzen wir PUT zum Aktualisieren if (hubBinId && !hubBinId.startsWith('hub_')) { endpoint = NPOINT_API_BASE + hubBinId; method = 'PUT'; } let response = await fetch(endpoint, { method: method, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); if (response.ok) { const result = await response.json(); // npoint.io gibt bei einem erfolgreichen POST die ID zurück (oft als .id oder direkt im Pfad) const newId = result.id || result.binId; if (newId) { hubBinId = newId; localStorage.setItem('active_hub_bin_id', hubBinId); window.history.replaceState({}, '', window.location.pathname + "?id=" + hubBinId); alert("✅ Erfolgreich in der npoint.io Cloud gespeichert! ID: " + hubBinId); await ladeDatenAusCloud(); return; } } // Alternativer npoint-Erstellungsversuch über Standard-Header, falls der erste fehlschlägt let fallbackResponse = await fetch("https://api.npoint.io", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) }); if (fallbackResponse.ok) { const fallbackResult = await fallbackResponse.json(); if (fallbackResult.id) { hubBinId = fallbackResult.id; localStorage.setItem('active_hub_bin_id', hubBinId); window.history.replaceState({}, '', window.location.pathname + "?id=" + hubBinId); alert("✅ Cloud-Speicher erfolgreich initialisiert! ID: " + hubBinId); await ladeDatenAusCloud(); return; } } throw new Error("npoint.io hat keine Bin-ID zurückgegeben."); } catch (e) { console.warn("Cloud-Speicherung fehlgeschlagen, nutze lokalen Container.", e); if (!hubBinId || hubBinId.startsWith('hub_')) { hubBinId = "hub_" + Math.random().toString(36).substring(2, 9); localStorage.setItem('active_hub_bin_id', hubBinId); window.history.replaceState({}, '', window.location.pathname + "?id=" + hubBinId); } localStorage.setItem(hubBinId, JSON.stringify(payload)); alert("⚠️ Cloud-Verbindung blockiert (500). Daten wurden sicher im lokalen Browser-Container gesichert: " + hubBinId); sammleUndZeigeTermine({ events: alleTermine }); } }
