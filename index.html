async function speichereInCloud() { const alleTermine = sammleTermineAusAllenTools(); const payload = { erstelltAm: new Date().toISOString(), events: alleTermine }; try { let endpoint = NPOINT_API_BASE; let method = 'POST'; // Falls bereits eine echte ID existiert, updaten wir sie (PUT) if (hubBinId && !hubBinId.startsWith('hub_')) { endpoint = NPOINT_API_BASE + hubBinId; method = 'PUT'; } let response = await fetch(endpoint, { method: method, headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' }, body: JSON.stringify(payload) }); if (response.ok) { const result = await response.json(); // npoint.io liefert bei einem erfolgreichen POST die neue ID im 'id'-Feld zurück const newId = result.id || result.binId; if (newId) { hubBinId = newId; localStorage.setItem('active_hub_bin_id', hubBinId); window.history.replaceState({}, '', window.location.pathname + "?id=" + hubBinId); alert("✅ Erfolgreich in der npoint.io Cloud gespeichert! ID: " + hubBinId); await ladeDatenAusCloud(); return; } } // Falls npoint.io muckt, sauberer Fallback auf lokal, damit es nicht hängt throw new Error("Server antwortete nicht mit einer ID."); } catch (e) { console.warn("Cloud-Speicherung direkt fehlgeschlagen, nutze lokalen Container.", e); if (!hubBinId || hubBinId.startsWith('hub_')) { hubBinId = "hub_" + Math.random().toString(36).substring(2, 9); localStorage.setItem('active_hub_bin_id', hubBinId); window.history.replaceState({}, '', window.location.pathname + "?id=" + hubBinId); } localStorage.setItem(hubBinId, JSON.stringify(payload)); alert("⚠️ npoint.io hat den Request abgewiesen (500). Daten wurden sicher im lokalen Browser-Container gesichert: " + hubBinId); sammleUndZeigeTermine({ events: alleTermine }); } }
